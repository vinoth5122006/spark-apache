19. cat input.txt

20. cat > wordcount.scala <<'EOF'
val input = sc.textFile("input.txt")
val count = input
  .flatMap(line => line.split("\\s+"))
  .filter(word => word.nonEmpty)
  .map(word => (word, 1))
  .reduceByKey(_ + _)
count.saveAsTextFile("output")
println("Word Count completed successfully!")
EOF

21. cat wordcount.scala

22. spark-shell -i wordcount.scala

23. :quit
